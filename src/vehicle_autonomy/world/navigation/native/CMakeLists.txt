cmake_minimum_required(VERSION 3.16)
project(vaf_native_source_receiver)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
add_executable(vaf_native_source_receiver source_receiver.cpp)
target_compile_features(vaf_native_source_receiver PRIVATE cxx_std_17)
ament_target_dependencies(vaf_native_source_receiver rclcpp geometry_msgs)
install(TARGETS vaf_native_source_receiver DESTINATION bin)
if(DEFINED ENV{VAF_ROS2_ROOT})
  target_link_directories(vaf_native_source_receiver PRIVATE "$ENV{VAF_ROS2_ROOT}/usr/lib/${CMAKE_LIBRARY_ARCHITECTURE}")
endif()
